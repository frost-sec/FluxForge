cmake_minimum_required(VERSION 3.22)
project(fluxforge_runner VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
find_package(Threads REQUIRED)
find_package(Protobuf REQUIRED)
set(WORKER_PROTO_OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${WORKER_PROTO_OUTPUT})
add_custom_command(OUTPUT ${WORKER_PROTO_OUTPUT}/worker.pb.cc ${WORKER_PROTO_OUTPUT}/worker.pb.h
  COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --cpp_out=${WORKER_PROTO_OUTPUT} ${CMAKE_CURRENT_SOURCE_DIR}/proto/worker.proto
  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/proto/worker.proto protobuf::protoc VERBATIM)
add_library(ff_worker_proto ${WORKER_PROTO_OUTPUT}/worker.pb.cc)
target_include_directories(ff_worker_proto PUBLIC ${WORKER_PROTO_OUTPUT})
target_link_libraries(ff_worker_proto PUBLIC protobuf::libprotobuf)
find_package(nlohmann_json 3.8 REQUIRED)
find_package(nlohmann_json_schema_validator QUIET)
if(NOT TARGET nlohmann_json_schema_validator::validator)
  include(FetchContent)
  set(JSON_VALIDATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(JSON_VALIDATOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(JSON_VALIDATOR_INSTALL OFF CACHE BOOL "" FORCE)
  set(JSON_VALIDATOR_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(json_schema_validator
    GIT_REPOSITORY https://github.com/pboettch/json-schema-validator.git
    GIT_TAG 2.3.0 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(json_schema_validator)
endif()
add_library(ff_common INTERFACE)
target_include_directories(ff_common INTERFACE include ../sdk/cpp/include)
target_link_libraries(ff_common INTERFACE nlohmann_json::nlohmann_json Threads::Threads ff_worker_proto)
add_executable(fluxforge-worker src/worker/main.cpp)
target_link_libraries(fluxforge-worker PRIVATE ff_common nlohmann_json_schema_validator::validator ${CMAKE_DL_LIBS})
add_executable(fluxforge-worker-host src/worker_host/main.cpp)
target_link_libraries(fluxforge-worker-host PRIVATE ff_common)
foreach(example echo add fan_in spin crash log_flood large_output delay)
  add_library(ff_${example} SHARED ../examples/functions/${example}.cpp)
  target_link_libraries(ff_${example} PRIVATE ff_common)
  set_target_properties(ff_${example} PROPERTIES PREFIX "" OUTPUT_NAME "${example}" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/examples")
endforeach()
option(FF_BUILD_AGENT "Build the gRPC Runner agent" ON)
if(FF_BUILD_AGENT)
  find_package(Protobuf REQUIRED)
  find_package(gRPC CONFIG QUIET)
  if(NOT TARGET gRPC::grpc++)
    # Ubuntu 22.04's distro gRPC packages expose pkg-config, not gRPCConfig.cmake.
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GRPC REQUIRED IMPORTED_TARGET grpc++)
    add_library(gRPC::grpc++ ALIAS PkgConfig::GRPC)
    find_program(GRPC_CPP_PLUGIN grpc_cpp_plugin REQUIRED)
    add_executable(gRPC::grpc_cpp_plugin IMPORTED GLOBAL)
    set_target_properties(gRPC::grpc_cpp_plugin PROPERTIES IMPORTED_LOCATION "${GRPC_CPP_PLUGIN}")
  endif()
  find_package(CURL REQUIRED)
  find_package(OpenSSL REQUIRED)
  set(PROTO_INPUT ${CMAKE_CURRENT_SOURCE_DIR}/../proto/fluxforge_v1.proto)
  set(PROTO_OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/generated)
  file(MAKE_DIRECTORY ${PROTO_OUTPUT})
  add_custom_command(OUTPUT ${PROTO_OUTPUT}/fluxforge_v1.pb.cc ${PROTO_OUTPUT}/fluxforge_v1.pb.h ${PROTO_OUTPUT}/fluxforge_v1.grpc.pb.cc ${PROTO_OUTPUT}/fluxforge_v1.grpc.pb.h
    COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/../proto --cpp_out=${PROTO_OUTPUT} --grpc_out=${PROTO_OUTPUT} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO_INPUT}
    DEPENDS ${PROTO_INPUT} protobuf::protoc gRPC::grpc_cpp_plugin VERBATIM)
  add_library(ff_proto ${PROTO_OUTPUT}/fluxforge_v1.pb.cc ${PROTO_OUTPUT}/fluxforge_v1.grpc.pb.cc)
  target_compile_options(ff_proto PRIVATE -Wno-deprecated-declarations)
  target_include_directories(ff_proto PUBLIC ${PROTO_OUTPUT})
  target_link_libraries(ff_proto PUBLIC gRPC::grpc++ protobuf::libprotobuf)
  add_executable(fluxforge_runner src/main.cpp)
  target_link_libraries(fluxforge_runner PRIVATE ff_common ff_proto CURL::libcurl OpenSSL::Crypto)
  install(TARGETS fluxforge_runner RUNTIME DESTINATION bin)
endif()
install(TARGETS fluxforge-worker fluxforge-worker-host RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(ff_ipc_test tests/ipc_test.cpp)
  target_link_libraries(ff_ipc_test PRIVATE ff_common)
  add_test(NAME ipc_framing COMMAND ff_ipc_test)
  add_executable(ff_worker_test tests/worker_test.cpp)
  target_link_libraries(ff_worker_test PRIVATE ff_common)
  add_test(NAME worker_lifecycle COMMAND ff_worker_test $<TARGET_FILE:fluxforge-worker-host> $<TARGET_FILE:fluxforge-worker> $<TARGET_FILE:ff_echo> $<TARGET_FILE:ff_spin> $<TARGET_FILE:ff_crash> $<TARGET_FILE:ff_log_flood> $<TARGET_FILE:ff_large_output>)
  set_tests_properties(worker_lifecycle PROPERTIES TIMEOUT 30)
endif()
